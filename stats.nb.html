mixed effect model of the effects of presence of wh-licensor, presence of gap, gap type and embedding level on surprisals from gpt2

library(tidyverse)

full_surprisals.gpt.object <- read_csv("data/model_outputs/gpt2_wilcox2022_embed4_object_gap_stimuli_with_surprisals.csv")

full_surprisals.gpt.subject <- read_csv("data/model_outputs/gpt2_wilcox2022_embed4_object_gap_stimuli_with_surprisals.csv")

surprisals_df.gpt.object <- data.frame(
    "id" = full_surprisals.gpt.object$sentence_group, 
    "wh" = full_surprisals.gpt.object$wh, 
    "gap" = full_surprisals.gpt.object$gap, 
    "condition" = full_surprisals.gpt.object$condition,
    "embed" = full_surprisals.gpt.object$levels_of_embedding, 
    "surprisal" = full_surprisals.gpt.object$local_surprisal,
    "gap_type" = "object")

surprisals_df.gpt.subject <- data.frame(
  "id" = full_surprisals.gpt.subject$sentence_group, 
  "wh" = full_surprisals.gpt.subject$wh, 
  "gap" = full_surprisals.gpt.subject$gap, 
  "condition" = full_surprisals.gpt.subject$condition,
  "embed" = full_surprisals.gpt.subject$levels_of_embedding, 
  "surprisal" = full_surprisals.gpt.subject$local_surprisal,
  "gap_type" = "subject"  
)

surprisals_df.gpt2 <- rbind(surprisals_df.gpt.subject, surprisals_df.gpt.object)
